  <script>

    // ★ご自身のGAS WebアプリURLを設定してください

    const GAS_URL = https://enoy2006.github.io/onsen-shot/


    let resultData = null;


    window.onload = async () => {

      try {

        const res = await fetch(GAS_URL);

        resultData = await res.json();

        document.getElementById("di-display").innerHTML = `<b>本日エリアの不快指数：${resultData.di}</b><br><small>不快指数高めにつき確率UP中！</small>`;

      } catch (e) {

        document.getElementById("di-display").innerText = "データの読み込みに失敗しました";

      }

    };


    function startSlot() {

      if (!resultData) return;

      document.getElementById("spin-btn").disabled = true;


      const reels = [document.getElementById("r1"), document.getElementById("r2"), document.getElementById("r3")];

      const targetSymbol = resultData.isWin ? "🫧" : "❌";


      reels.forEach((reel, i) => {

        setTimeout(() => {

          reel.innerText = targetSymbol;

          if (i === 2) finishGame();

        }, (i + 1) * 600);

      });

    }


    function finishGame() {

      if (resultData.isWin) {

        document.getElementById("coupon-title").innerText = resultData.couponName;

        document.getElementById("coupon-card").style.display = "block";

      } else {

        alert("残念！また明日チャレンジしてね！");

      }

    }

  </script>
